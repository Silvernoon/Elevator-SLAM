option(VINS_BUILD_YOLO "Build the standalone Darknet detector" ON)
option(VINS_YOLO_CUDA "Use CUDA in the standalone detector" OFF)
if(VINS_BUILD_YOLO)
  find_package(Boost REQUIRED COMPONENTS thread)
  set(DARKNET_PATH ${CMAKE_CURRENT_SOURCE_DIR}/third_party/darknet)
  file(GLOB DARKNET_C_SOURCES CONFIGURE_DEPENDS ${DARKNET_PATH}/src/*.c)
  list(REMOVE_ITEM DARKNET_C_SOURCES ${DARKNET_PATH}/src/compare.c)
  if(VINS_YOLO_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    file(GLOB DARKNET_CUDA_SOURCES ${DARKNET_PATH}/src/*.cu)
    add_library(vins_darknet ${DARKNET_C_SOURCES} ${DARKNET_PATH}/src/image_opencv.cpp ${DARKNET_CUDA_SOURCES})
    target_compile_definitions(vins_darknet PUBLIC GPU)
    target_link_libraries(vins_darknet PUBLIC CUDA::cudart CUDA::cublas CUDA::curand)
  else()
    add_library(vins_darknet ${DARKNET_C_SOURCES} ${DARKNET_PATH}/src/image_opencv.cpp)
  endif()
  target_compile_definitions(vins_darknet PUBLIC OPENCV)
  target_include_directories(vins_darknet PUBLIC ${DARKNET_PATH}/src ${DARKNET_PATH}/include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(vins_darknet PUBLIC ${OpenCV_LIBS} Threads::Threads m)
  add_library(darknet_ros_lib src/YoloObjectDetector.cpp src/image_interface.cpp)
  target_include_directories(darknet_ros_lib PUBLIC include ${Boost_INCLUDE_DIRS})
  ament_target_dependencies(darknet_ros_lib PUBLIC rclcpp rclcpp_action image_transport cv_bridge sensor_msgs std_msgs darknet_ros_msgs)
  target_link_libraries(darknet_ros_lib PUBLIC vins_darknet ${Boost_LIBRARIES})
  add_executable(darknet_ros src/yolo_object_detector_node.cpp)
  ament_target_dependencies(darknet_ros PUBLIC rclcpp rclcpp_action image_transport cv_bridge sensor_msgs std_msgs darknet_ros_msgs)
  target_link_libraries(darknet_ros PRIVATE darknet_ros_lib)
  install(TARGETS vins_darknet darknet_ros_lib darknet_ros ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
endif()

install(TARGETS vins_estimator vins_node vins_line_descriptor ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
